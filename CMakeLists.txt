cmake_minimum_required(VERSION 3.20)
project(GH2PC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
file(GLOB GH2_TRANSLATION CONFIGURE_DEPENDS "generated/ppc_recomp.*.cpp")
add_library(gh2_code STATIC ${GH2_TRANSLATION} generated/ppc_func_mapping.cpp)
target_include_directories(gh2_code PUBLIC generated runtime tools/XenonRecomp/thirdparty/simde)
target_compile_options(gh2_code PRIVATE -msse4.1 -mavx -fno-strict-aliasing)
target_compile_definitions(gh2_code PUBLIC GH2_TRACE_ENTRY)
add_executable(gh2_bootstrap runtime/bootstrap.cpp runtime/kernel_runtime.cpp runtime/game_files.cpp runtime/guest_clock.cpp generated/gh2_import_stubs.cpp)
target_link_libraries(gh2_bootstrap PRIVATE gh2_code)
target_compile_options(gh2_bootstrap PRIVATE -msse4.1 -mavx)
target_link_options(gh2_bootstrap PRIVATE -static)
